#include <stdio.h>
#include <string.h>
int main()
{
    char name[255];
    printf("Введите Ваше имя: ");
    fgets(name, 255, stdin);
    printf("length = %d\n", strlen(name)); /* debug line */
    name[strlen(name)-1] = '\0'; /* remove the newline at the end */
    printf("Привет %s!\n", name);
    return 0;
}
